## Замер такта физики для rpg_perf (пока Prof копит; иначе — пустой вызов): первый узел дерева по
## process_physics_priority — начало, последний — конец всех _physics_process (мод и игра, деревья анимаций в такте
## физики). Таймер физики, заведённый последним, срабатывает после очереди отложенных вызовов того же такта (поза
## скелетов, модификаторы — IK стоп, поворот головы). Первый _process кадра, где прошёл ровно один такт, — после
## шага сервера физики и уведомлений о трансформах: «хвост» такта и такт целиком (≈, с началом кадра).
class TickProbe extends Node:
	const P := preload("res://mods/pax_rpg/src/location/prof.gd")
	static var t0 := 0                   # начало прохода узлов такта
	static var t_end := 0                # после отложенных вызовов такта
	var last := false
	var _pf := -1

	func _physics_process(_delta: float) -> void:
		if not P.on:
			return
		if not last:
			t0 = Time.get_ticks_usec()
			return
		P.add("такт: все _physics_process", t0)
		P.peak("такт: все _physics_process", t0)
		get_tree().create_timer(0.0, true, true).timeout.connect(_after)

	func _after() -> void:
		t_end = Time.get_ticks_usec()
		P.add("такт: с отложенными (скелеты)", t0)
		P.peak("такт: с отложенными (скелеты)", t0)

	func _process(_delta: float) -> void:
		if last or not P.on:
			return
		var pf := Engine.get_physics_frames()
		if pf == _pf + 1 and t_end > t0:
			P.add("такт: шаг сервера и уведомления (≈)", t_end)
			P.peak("такт: шаг сервера и уведомления (≈)", t_end)
			P.peak("такт целиком (≈)", t0)
		_pf = pf


func _f(key: String, fallback: float) -> float:
	return float(cfg.get(key, fallback))

## Гоблины локации (config/location.json → goblins): полукругом перед героем, между ним и воротами.
func _spawn_goblins(n: int) -> void:
	for i: int in n:
		var g := Goblin.new()
		var a := (float(i) - (n - 1) * 0.5) * 0.6
		g.setup(mod, _player, _on_hero_hit, START + Vector3(sin(a) * 9.0, -1.0, -cos(a) * 9.0), _hero_attacking,
			_hero_blocking)
		_view.add_child(g)
		_goblins.append(g)

func spawn_orc() -> String:
	if _view == null or not is_instance_valid(_player):
		return "нет локации"
	var g := Orc.new()
	var fwd := Vector3(-sin(_body_mesh.rotation.y), 0.0, -cos(_body_mesh.rotation.y))
	g.setup(mod, _player, _on_hero_hit, _player.global_position + fwd * 8.0, _hero_attacking, _hero_blocking)
	g.slammed.connect(_orc_slam)
	_view.add_child(g)
	_goblins.append(g)
	return "orc spawned"


func _hp_max() -> float:
	return HERO_HP * (_prog.hp_mult() if _prog != null else 1.0)


func _conc_max() -> float:
	return HERO_CONC * (_prog.posture_mult() if _prog != null else 1.0)


## Герой беспомощен (слом концентрации или миг оглушения): ни ударов, ни приёмов, ни эликсира.
func _helpless() -> bool:
	return _stagger > 0.0 or _broken_t > 0.0


## Трата концентрации героя; дошла до 0 — слом (by — кто сломал: он и казнит).
func _spend(a: float, by: Node3D = null) -> void:
	if a <= 0.0 or _broken_t > 0.0:
		return
	_conc = maxf(0.0, _conc - a)
	_conc_idle = 0.0
	if _conc <= 0.0:
		_hero_break(by)


## Прибавка концентрации (доля от полной): ульта, попавшая всеми ударами, — +50 %.
func _conc_gain(frac: float) -> void:
	if _broken_t > 0.0:
		return
	_conc = minf(_conc_max(), _conc + _conc_max() * frac)


## Слом героя: шатается и падает на колено (клип Stagger), BROKEN_TIME с без управления — враги спешат казнить
## (goblin.gd → hero_broken); удар врага в это время — Кровожадная атака по герою: треть здоровья.
func _hero_break(_by: Node3D) -> void:
	_conc = 0.0
	_broken_t = BROKEN_TIME
	_rmb = false
	if _hero != null:
		_hero.set_block(false)
		_hero.stagger()
	_toast = {"text": mod.tr_key("pax_rpg_conc_broken"), "t": 0.0}
	_sfx_once("sounds/body_fall.ogg", -6.0)
	_flash_t = maxf(_flash_t, _hf("fx_flash_time", 0.22))


## Готова ли локация (экран загрузки не открывает её раньше — travel.gd).
func is_ready() -> bool:
	return not building


## Заранее, ещё на глобусе: горы и степь городов считаются в потоке — к входу в город уже готовы.
func prewarm() -> void:
	var parsed: Variant = JSON.parse_string(mod.read_text("config/cities.json"))
	if not (parsed is Dictionary):
		return
	for c: Variant in (parsed as Dictionary).get("cities", []):

## 3D игры (глобус, космос) под локацией не видно, но оно рисовалось каждый кадр — пока герой в городе, окно игры
## не рисует свой 3D (свой мир локации — в своём SubViewport и рисуется дальше).
func _game_3d(on: bool) -> void:
	var root := get_tree().root
	if root != null:
		root.disable_3d = not on
	_game_2d(on)


## То же с 2D: окна и панели игры и главное меню (слои окна ниже нашего, 100) закрыты непрозрачным 3D-видом, но рисовались
## каждый кадр — полноэкранный шейдер фона меню, панель слоёв карты, которую игра открывает по TAB (+0.5 мс видеокарты).
## Пока герой в городе, они спрятаны; новые, что игра откроет под нами, прячутся раз в 0.5 с (_under_t); на выходе
## возвращаются только те, что спрятали мы. Слои выше нашего (консоль, экран загрузки) не трогаем — их видно.
var _hidden_2d: Array = []
var _under_t := 0.0


func _game_2d(on: bool) -> void:
	if on:
		for n: Variant in _hidden_2d:
			if is_instance_valid(n):
				(n as CanvasLayer).visible = true
		_hidden_2d.clear()
		return
	_hide_under()


func _hide_under() -> void:
	var root := get_tree().root
	for n: Node in root.find_children("*", "CanvasLayer", true, false):
		var cl := n as CanvasLayer
		if cl == self or not cl.visible or cl.layer >= layer or cl.get_viewport() != root or is_ancestor_of(cl):
			continue
		cl.visible = false
		_hidden_2d.append(cl)


## Всё построено и трава встала — локацию можно показывать.
func _check_ready(gen: int) -> void:
	if gen == _build_gen and _build_finished and (_grass == null or _grass.done) and building:
		building = false
		_mark("ready (grass %d)" % (_grass.count if _grass != null else 0))
		get_tree().create_timer(0.5).timeout.connect(func() -> void:   # облака, блики, размытие — уже после загрузки
			if gen == _build_gen and is_instance_valid(_gfx):
				_gfx.apply_camera(_cam, _sun))


## Журнал загрузки (godot.log): этап и секунды от начала — видно, где загрузка теряет время.
var _load_t0 := 0


func _mark(stage: String) -> void:
	print("[pax_rpg] load %.2f s: %s" % [(Time.get_ticks_msec() - _load_t0) / 1000.0, stage])


## Прогрев шейдеров: эффекты, которые впервые появятся только в бою (искры со шлейфом, вспышка, лужа крови, разрез
## и рыбий глаз иая), один раз рисуются перед камерой, пока экран загрузки закрыт, — их шейдеры и конвейеры
## компилируются сейчас, а не рывком на первом ударе.
func _warmup() -> void:
	if not is_instance_valid(_cam) or not is_instance_valid(_view):
		return
	var at := _cam.global_position - _cam.global_transform.basis.z * 3.0
	_sparks(at, Color(1.0, 0.8, 0.35), 4, 1.0)
	Goblin.blood_decal(_view, at - Vector3(0.0, 1.0, 0.0))
	Rift.warm(mod, _view, at - Vector3(0.0, 1.0, 0.0))   # «Разлом»: трещины, шипы, их шейдеры — сейчас, а не рывком в бою
	if is_instance_valid(_iai):
		_iai.warmup(at)

func _make_sky() -> void:

	_sky_mat = null
	if ssh != null:                                  # небо со сменой суток и облаками (daynight.gd)
		_sky_mat = ShaderMaterial.new()

		sky.sky_material = _sky_mat
		sky.process_mode = Sky.PROCESS_MODE_INCREMENTAL
		sky.radiance_size = Sky.RADIANCE_SIZE_64
	else:
		var sky_mat := ProceduralSkyMaterial.new()
		sky_mat.sky_top_color = Color(0.22, 0.42, 0.85)
		sky_mat.sky_horizon_color = Color(1.0, 0.62, 0.86)     # розовая мановая атмосфера Юрумы
		sky_mat.ground_horizon_color = Color(0.85, 0.55, 0.8)
		sky_mat.ground_bottom_color = Color(0.1, 0.12, 0.3)
		sky.sky_material = sky_mat
	var env := Environment.new()
	env.background_mode = Environment.BG_SKY
	env.sky = sky
	env.ambient_light_source = Environment.AMBIENT_SOURCE_SKY

func _sink_arena(layout: Dictionary) -> void:
	var ar: Array = layout.get("arena", [])

## Ко дворцу хана — лестница: цоколь (коробка 52 × 3 × 36 у z = −54) раньше стоял глухой стеной перед входом.
## Ступени — для вида (без коллизии), под ними пологий пандус с коллизией: герой поднимается шагом.
func _palace_stairs(layout: Dictionary) -> void:
	var items: Array = layout.get("items", [])
	for it: Variant in items:
		var d: Dictionary = it
		if d.has("box") and str(d.get("mat", "")) == "stone_cut":
			var b: Array = d.box
			var p: Array = d.p
			if float(b[1]) >= 2.0 and float(b[0]) > 30.0:
				var h := float(b[1])
				var front := float(p[2]) + float(b[2]) * 0.5     # к воротам (+Z)
				var run := h * 3.2                               # длина лестницы (уклон ~17°)
				var steps := int(ceil(h / 0.25))
				# старый пандус планировки (наклонная плита и площадка) стоял там же — две лестницы друг в друге
				var keep: Array = []
				for o: Variant in items:
					var op: Array = (o as Dictionary).get("p", [0, 0, 0])
					var inside: bool = (o as Dictionary).has("box") and o != it and absf(float(op[0]) - float(p[0])) < 8.0 \
						and float(op[2]) > front - 1.0 and float(op[2]) < front + run + 1.0
					if not inside:
						keep.append(o)
				items = keep
				for i: int in steps:
					var top := h - i * (h / steps)
					var dz := run / steps
					items.append({"box": [14, top, dz], "mat": "stone_cut", "p": [float(p[0]), top * 0.5, front + dz * (i + 0.5)],
						"r": 0.0, "c": false})
				var slope := atan2(h, run)
				items.append({"box": [14, 0.2, sqrt(h * h + run * run)], "mat": "stone_cut",
					"p": [float(p[0]), h * 0.5 - 0.05, front + run * 0.5], "r": 0.0, "rx": rad_to_deg(slope), "c": true, "ramp": true})
				break
	layout["items"] = items

## Ближайший житель в 2.5 м — подсказка над ним (раз в 0.2 с).
func _talk_scan(delta: float) -> void:
	_talk_t -= delta
	if _talk_t > 0.0 or not is_instance_valid(_player):
		return
	_talk_t = 0.2
	var best: Node = null
	var bd := 2.5
	for n: Variant in _npcs:
		if not is_instance_valid(n):
			continue
		var d := (n as Node3D).global_position.distance_to(_player.global_position)
		if d < bd:
			bd = d
			best = n
	if best != _talk_npc:
		if is_instance_valid(_talk_npc):
			_talk_npc.call("show_prompt", false)
		_talk_npc = best
	if is_instance_valid(_talk_npc) and not (is_instance_valid(_talk) and _talk.visible):
		_talk_npc.call("show_prompt", true)


func _on_voice(text: String) -> void:

	if hit.is_empty():
		_vtalk.hint(mod.tr_key("pax_rpg_voice_nobody"))

		_talk.say_text(str(hit[1]))

## Печатает ли игрок (фокус в поле ввода): ответ — один раз на кадр, а зовут его десятки раз (ввод, клавиши).
func _typing() -> bool:

		_typing_val = _typing_now()
	return _typing_val


func _typing_now() -> bool:
	var focus := get_viewport().gui_get_focus_owner()
	# фокус на скрытом поле (окно свернули во время загрузки — поле ввода игры или имени осталось в фокусе
	# невидимым) — не печать: иначе клавиши героя молчат, работает одно колесо мыши
	if (focus is LineEdit or focus is TextEdit) and not focus.is_visible_in_tree():
		get_viewport().gui_release_focus()
		return false
	return focus is LineEdit or focus is TextEdit

## Финал Иай: удар по каждому, кого задели разрезы. Рядовые падают, у босса — тяжёлый урон и сбитая стойка.
func _iai_cut(v: Node) -> void:
	var g := v as Goblin
	if g == null or not g.is_alive():
		return
	var dmg := _hf("iai_damage", 60.0) * _dmg_mult()
	if not g.boss and is_instance_valid(_cam):          # разрез — по диагонали кадра: тело рубится пополам
		var cb := _cam.global_transform.basis
		var line := cb.x.rotated(-cb.z, randf_range(-1.0, 1.0))
		g.mark_slice((-cb.z).cross(line))
	g.take_damage(dmg if g.boss else maxf(dmg, g.hp + 1.0), _player.global_position, not g.boss)
	if g.boss:
		g.spend(g.conc_max * 0.4)                        # босс: выдержка сбита сильно
	elif g.is_alive():
		g.deathblow()                                    # щелчок цубы — падает сразу, без «на коленях»
	_iai_hits += 1
	_numbers.append({"pos": g.global_position + Vector3(0, 1.4, 0), "v": int(dmg), "crit": true, "t": 0.0})

## Удар ультимейта о землю: перед героем; 1 — первая волна, 2 — выпад: урон вдвое, площадь шире.
func _ult_wave(i: int) -> void:
	var fwd := Vector3(-sin(_body_mesh.rotation.y), 0.0, -cos(_body_mesh.rotation.y))
	var at := _player.global_position + fwd * (1.8 if i == 1 else 2.4)
	var rings: Array = [[3.0, 30.0], [5.0, 22.0]] if i == 1 else [[3.5, 60.0], [6.5, 45.0], [9.5, 35.0]]
	if bool(mod.get_setting("combat_fx", true)):          # волна, корона пыли, вспышка у земли (hit_vfx.gd)
		var col: Array = hero_cfg.get("mana_color", hero_cfg.get("trail_color", [0.8, 0.45, 1.0]))
		var cc := Color(float(col[0]), float(col[1]), float(col[2]))
		_landing_fx().landing(at - Vector3(0.0, 0.9, 0.0), cc, 3.0 if i == 1 else 5.0)
		_screen_hit(at - Vector3(0.0, 0.9, 0.0), cc, 0.6 if i == 1 else 1.0)
	for k: int in rings.size():
		var r: Array = rings[k]
		var tw := get_tree().create_timer(0.12 * k)
		tw.timeout.connect(_quake_wave.bind(float(r[0]), float(r[1]), 2 if (i == 2 and k == rings.size() - 1) else k, at))
	_shake = maxf(_shake, 0.6 if i == 1 else 1.0)

func _quake_step() -> void:

		return
	var waves: Array = hero_cfg.get("quake_waves", [[3.0, 22.0], [5.5, 18.0], [8.0, 15.0]])

		var w: Array = waves[_quake_waves_done]
		_quake_wave(float(w[0]), float(w[1]), _quake_waves_done)
		_quake_waves_done += 1


var _quake_hit_waves := 0
var _iai_hits := 0


func _quake_wave(radius: float, dmg: float, i: int, at := Vector3.INF) -> void:
	var c := (_player.global_position if at == Vector3.INF else at) - Vector3(0, 0.9, 0)
	var hit_any := false
	for g: Variant in _goblins:
		if not is_instance_valid(g) or not (g as Goblin).is_alive():
			continue
		var gob: Goblin = g
		var d := Vector2(gob.global_position.x - c.x, gob.global_position.z - c.z).length()
		if d <= radius:
			gob.take_damage(dmg * _dmg_mult(), c, i == 2)          # последняя волна — отбрасывает
			_style.hit("Ultimate" if _hero != null and _hero.action == "Ultimate" else "Quake")
			gob.spend(dmg * 0.8)
			hit_any = true
			if d >= maxf(0.0, radius - 3.0):                               # шипы на кольце волны — подбрасывают
				var away := Vector3(gob.global_position.x - c.x, 0.0, gob.global_position.z - c.z).normalized()
				gob.launch(away * 2.5 + Vector3(0.0, 7.5 if not gob.boss else 3.5, 0.0))
			_numbers.append({"pos": gob.global_position + Vector3(0, 1.3, 0), "v": int(dmg), "crit": i == 2, "t": 0.0})
	if hit_any and _hero != null and _hero.action == "Quake":
		_quake_hit_waves += 1
		var nw: Array = hero_cfg.get("quake_waves", [[3.0, 22.0], [5.5, 18.0], [8.0, 15.0]])
		if _quake_hit_waves == nw.size():
			_conc_gain(0.5)                              # разлом задел врагов всеми волнами — +50 % концентрации
	if _sfx != null:
		_sfx.impact()
	# «Разлом» Разрубателя: земля раскалывается (декаль трещин с глубиной), по кольцу волны — земляные шипы
	var waves: Array = hero_cfg.get("quake_waves", [[3.0, 22.0], [5.5, 18.0], [8.0, 15.0]])
	if i == 0:
		Rift.crack(mod, _view, c, float((waves[waves.size() - 1] as Array)[0]) + 1.5, 7.0)
	# зона растёт понемногу: шипы и комья земли идут кольцами от прошлой волны к этой (шаг ~0.8 м, 70 мс)
	var r0 := maxf(0.9, radius - 2.5)
	var steps := maxi(1, int(ceil((radius - r0) / 0.8)))
	for k: int in steps:
		var ra := r0 + (radius - r0) * float(k) / steps
		var rb := r0 + (radius - r0) * float(k + 1) / steps
		var t := get_tree().create_timer(0.07 * k)
		t.timeout.connect(func() -> void:
			if not is_instance_valid(_view):
				return
			Rift.spikes(mod, _view, c, ra, rb, maxi(2, int(rb * 2.2 / steps * 1.6)), 0.35)
			_quake_debris(c, rb))
	# волна: пыль по кругу (без светящегося кольца — земля трескается, а не светится)
	if _vfx == null:
		_vfx = Vfx.new(mod)
	var dust := _vfx.smoke(_view, c + Vector3(0, 0.2, 0), 0.9 + i * 0.3, Color(0.7, 0.58, 0.5, 0.5), 18, 1.4)
	dust.one_shot = true
	dust.explosiveness = 0.9
	dust.preprocess = 0.0
	dust.emission_shape = CPUParticles3D.EMISSION_SHAPE_RING
	dust.emission_ring_axis = Vector3.UP
	dust.emission_ring_radius = radius
	dust.emission_ring_inner_radius = radius * 0.8
	dust.emission_ring_height = 0.1
	dust.emitting = true
	dust.finished.connect(dust.queue_free)
	_sparks(c + Vector3(0, 0.3, 0), Color(0.62, 0.48, 0.36), 30 + i * 10, 6.0)   # каменная крошка, не мана

## Волна разреза (финишер, как в DMC): светящийся серп летит вперёд над землёй, режет всех на пути (раз каждого).
func _slash_wave() -> void:
	var fwd := Vector3(-sin(_body_mesh.rotation.y), 0.0, -cos(_body_mesh.rotation.y))
	var node := MeshInstance3D.new()
	if not _fx_res.has("arc"):                            # серп — одна сетка на все волны (материал — свой: гаснет)
		var st := SurfaceTool.new()
		st.begin(Mesh.PRIMITIVE_TRIANGLES)
		var n := 24
		for i: int in n:                                  # дуга 150° перед героем: толще в середине, тонко к краям
			var a0 := deg_to_rad(-75.0 + 150.0 * i / n)
			var a1 := deg_to_rad(-75.0 + 150.0 * (i + 1) / n)
			var w0 := 0.28 * sin(PI * float(i) / n) + 0.02
			var w1 := 0.28 * sin(PI * float(i + 1) / n) + 0.02
			var p0 := Vector3(sin(a0), 0.0, -cos(a0)) * 1.6
			var p1 := Vector3(sin(a1), 0.0, -cos(a1)) * 1.6
			var q0 := Vector3(sin(a0), 0.0, -cos(a0)) * (1.6 - w0)
			var q1 := Vector3(sin(a1), 0.0, -cos(a1)) * (1.6 - w1)
			for v: Vector3 in [p0, p1, q1, p0, q1, q0]:
				st.add_vertex(v)
		_fx_res["arc"] = st.commit()
	node.mesh = _fx_res["arc"]
	var mat := StandardMaterial3D.new()
	mat.shading_mode = BaseMaterial3D.SHADING_MODE_UNSHADED
	mat.transparency = BaseMaterial3D.TRANSPARENCY_ALPHA
	mat.blend_mode = BaseMaterial3D.BLEND_MODE_ADD
	mat.cull_mode = BaseMaterial3D.CULL_DISABLED
	mat.albedo_color = Color(0.75, 0.55, 1.6, 0.9)
	node.material_override = mat
	node.cast_shadow = GeometryInstance3D.SHADOW_CASTING_SETTING_OFF
	_view.add_child(node)
	node.global_position = _player.global_position + Vector3(0, 0.1, 0) + fwd * 0.6
	node.rotation.y = _body_mesh.rotation.y
	_waves.append({"node": node, "dir": fwd, "t": 0.0, "hit": {}, "mat": mat})
	_shake = _f("shake_time", 0.18)
	_shake_amp = _hf("shake_heavy", 0.09)
	_sfx_once("sounds/foe_swing_1.mp3", 2.0)


func _waves_tick(delta: float) -> void:
	for w: Dictionary in _waves.duplicate():
		var node: MeshInstance3D = w.node
		if not is_instance_valid(node):
			_waves.erase(w)
			continue
		w.t = float(w.t) + delta
		var k := float(w.t) / _hf("wave_time", 0.7)
		node.global_position += (w.dir as Vector3) * _hf("wave_speed", 16.0) * delta
		node.scale = Vector3.ONE * (1.0 + k * 1.2)
		(w.mat as StandardMaterial3D).albedo_color.a = 0.9 * (1.0 - k)
		var reach := 1.6 * node.scale.x
		for g: Variant in _goblins:
			if not is_instance_valid(g) or not (g as Goblin).is_alive() or (w.hit as Dictionary).has(g):
				continue
			var gob: Goblin = g
			var to := gob.global_position - node.global_position
			to.y = 0.0
			if to.length() < reach + 0.5 and to.normalized().dot(w.dir as Vector3) > -0.2:
				(w.hit as Dictionary)[g] = true
				var dmg := _hf("wave_damage", 40.0) * _dmg_mult()
				gob.take_damage(dmg, node.global_position, true)
				_style.hit("Finisher")
				_impact_feel(gob, true, 1.6)
				_numbers.append({"pos": gob.global_position + Vector3(0, 1.3, 0), "v": int(dmg), "crit": true, "t": 0.0})
		if k >= 1.0:
			node.queue_free()
			_waves.erase(w)


## Орк приземлился с прыжка: мини-землетрясение — трещины, кольцо шипов, комья земли, пыль, дрожь камеры.
func _orc_slam(at: Vector3) -> void:
	if not is_instance_valid(_view):
		return
	var c := Vector3(at.x, at.y, at.z)
	var d0 := c.distance_to(_player.global_position)
	Rift.crack(mod, _view, c, 3.2, 4.0)
	if bool(mod.get_setting("combat_fx", true)):          # волна, корона пыли, вспышка (hit_vfx.gd)
		_landing_fx().landing(c, Color(1.0, 0.42, 0.1), 3.2)
		_screen_hit(c, Color(1.0, 0.45, 0.15), clampf(1.4 - d0 / 14.0, 0.3, 1.0))
	for k: int in 3:                                      # растёт кольцами
		var ra := 0.6 + k * 0.8
		var t := get_tree().create_timer(0.07 * k)
		t.timeout.connect(func() -> void:
			if is_instance_valid(_view):
				Rift.spikes(mod, _view, c, ra, ra + 0.8, 5 + k * 3, 0.25)
				_quake_debris(c, ra + 0.8))
	_sparks(c + Vector3(0, 0.3, 0), Color(0.62, 0.48, 0.36), 30, 5.0)
	var d := c.distance_to(_player.global_position)
	if d < 9.0:
		_shake = _f("shake_time", 0.18) * 2.0
		_shake_amp = _hf("shake_heavy", 0.09) * (1.6 - d / 9.0)
	if _sfx != null:
		_sfx.impact()


func _foe_near(r: float) -> bool:

			return true
	return false

func _physics_body(delta: float) -> void:
	var tp := Time.get_ticks_usec()                      # метки частей такта (rpg_perf prof); без замера — пустые вызовы
	var on_floor := _player.is_on_floor()
	# F и пробел опрашиваем здесь, а не в _input: игра может забрать нажатие раньше нас
	var f := (Input.is_physical_key_pressed(KEY_F) or Input.is_key_pressed(KEY_F)) and not _typing()
	if (_f_queued or (f and not _f_down)) and _hero != null:
		var wasd := Input.is_physical_key_pressed(KEY_W) or Input.is_physical_key_pressed(KEY_A) \
			or Input.is_physical_key_pressed(KEY_S) or Input.is_physical_key_pressed(KEY_D)
		_hero.toggle_sword(on_floor, wasd)
	_f_queued = false
	_f_down = f
	_stagger = maxf(0.0, _stagger - delta)
	_since_hit += delta
	if _broken_t > 0.0:
		_broken_t = maxf(0.0, _broken_t - delta)
		if _broken_t <= 0.0:
			_conc = _conc_max() * _hf("conc_rise", 0.4)     # слом прошёл без казни — выдержка частично вернулась
	Goblin.hero_broken = _broken_t > 0.0
	_regen_conc(delta)
	if _hero != null:
		_hero.tired = _conc < _conc_max() / 3.0
	if _hero != null:
		_hero.combat = _hud_enemy() != null
		_hero.lock_on = _lock != null
	if _lock != null and (not is_instance_valid(_lock) or not (_lock as Goblin).is_alive()
			or _lock.global_position.distance_to(_player.global_position) > 22.0):
		_set_lock(null)
	if _lmb_queued and _hero != null and not _helpless():
		if not _try_bloodthirsty():
			var cy := _combo_yaw()
			_hero.swing_side = 0                     # враг сбоку: слева — взмах справа налево, справа — зеркально
			if cy != INF:
				var side := angle_difference(_body_mesh.rotation.y, cy)
				if absf(side) > deg_to_rad(_hf("swing_side_angle", 12.0)):
					_hero.swing_side = -1 if side > 0.0 else 1
			if cy != INF and (_hero.action == "" or _hero.action in Hero.ATTACKS):
				_face_now(cy)                        # удар — во врага в захвате, иначе по ходу (WASD); без ввода — куда стоит
			var fw := Input.is_physical_key_pressed(KEY_W)
			var bk := Input.is_physical_key_pressed(KEY_S)
			var done_sp := false
			if _lock != null and fw and not bk:          # захват + вперёд + ЛКМ — «Выпад»
				done_sp = _hero.special("Lunge", on_floor)
			elif _lock != null and bk and not fw:        # захват + назад + ЛКМ — «Подброс»
				done_sp = _hero.special("Launcher", on_floor)
			elif _hero.pause_finisher_ready():           # ЛКМ, ЛКМ, пауза, ЛКМ — «Пауза»
				done_sp = _hero.special("PauseFinisher", on_floor)
			if not done_sp:
				_hero.attack(on_floor)
	_lmb_queued = false
	if _hero != null:
		_hero.set_block(_rmb and not _helpless())
	var locked := _hero != null and (_hero.locks_move() or _helpless())
	var input := Vector2.ZERO
	if not _typing() and not locked:
		input = Vector2(float(Input.is_physical_key_pressed(KEY_D)) - float(Input.is_physical_key_pressed(KEY_A)),
			float(Input.is_physical_key_pressed(KEY_S)) - float(Input.is_physical_key_pressed(KEY_W)))
	var forward := Vector3(0.0, 0.0, -1.0).rotated(Vector3.UP, _yaw)
	var right := Vector3(1.0, 0.0, 0.0).rotated(Vector3.UP, _yaw)
	var move := right * input.x - forward * input.y
	if move.length() > 1.0:
		move = move.normalized()
	# уворот (Alt): развернуть героя по вводу (без ввода — куда смотрит) и кувырок; дальше его ведёт root motion
	var alt := Input.is_physical_key_pressed(KEY_ALT) and not _typing()
	if alt and not _alt_down and not locked and _hero != null:
		var wish := right * input.x - forward * input.y
		# в захвате — как в соулсах: лицом к врагу; назад — отскок, вбок — уклон, вперёд — кувырок
		var dir := ""
		if _lock == null and _hero.has("DodgeL"):         # без захвата — шаги относительно камеры: A/D — вбок, S или
			if wish.length() < 0.01 or input.y > 0.5:     # без ввода — назад, W — кувырок (как с захватом, только «лицом» — камера)
				dir = "B"
			elif absf(input.x) > absf(input.y):
				dir = "R" if input.x > 0.0 else "L"
			if dir != "":
				_face_now(_yaw)
		elif _lock != null and wish.length() > 0.01:
			var ty := _aim_yaw()
			var lf := Vector3(-sin(ty), 0.0, -cos(ty))
			var lr := Vector3(cos(ty), 0.0, -sin(ty))
			var w := wish.normalized()
			var fy := w.dot(lf)
			if fy < -0.5:
				dir = "B"
			elif fy < 0.6:
				dir = "R" if w.dot(lr) > 0.0 else "L"
			if dir != "":
				_face_now(ty)
		if _hero.dodge(on_floor, dir):
			if is_instance_valid(_sfx):
				_sfx.cloth(-4.0)
			if _hero.side_dodge():
				pass                                     # уклон — лицом к врагу, смещение ведёт root motion клипа
			elif wish.length() > 0.01:
				_body_mesh.rotation.y = atan2(-wish.x, -wish.z)
			locked = true
	_alt_down = alt
	var running := Input.is_physical_key_pressed(KEY_SHIFT) and not (_hero != null and (_hero.walk_only()
		or _hero.blocking))
	var speed := _hf("run_speed", 5.0) if running else _hf("walk_speed", 2.4)
	if _hero != null and (_hero.armed or _hero.walk_only()):
		speed *= _hf("armed_run_factor", 0.8) if running else _hf("armed_speed_factor", 1.0)
	if _hero != null and _hero.blocking:
		speed *= _hf("block_speed_factor", 0.55)
	elif _lock != null and not running:
		speed *= _hf("lock_speed_factor", 0.75)          # в поединке — осторожный шаг
	if _hero != null and not running and _hero.strafe_speed() >= 0.0:
		speed = _hero.strafe_speed()                     # шаг вбок: герой идёт так, как шагают ноги в клипе
	var vel := _player.velocity
	var want_v := Vector2(move.x, move.z) * speed
	var hv := _w.approach(Vector2(vel.x, vel.z), want_v, delta) if on_floor else want_v   # инерция: a = F/m, стоп — трением
	var es := _hero.edge_speed() if _hero != null and on_floor else -1.0
	if es >= 0.0:                                    # разгон / торможение бега — скорость из клипа, по лицу героя
		var fw := -_body_mesh.global_transform.basis.z
		var dir2 := Vector2(move.x, move.z).normalized() if move.length() > 0.01 else Vector2(fw.x, fw.z).normalized()
		hv = dir2 * es
	vel.x = hv.x

## Отладка (консоль rpg_probe): всё видимое в радиусе 2 м от героя — имя, родитель, высота над ступнями.
func probe() -> String:
	if not is_instance_valid(_player):
		return "нет героя"
	var out: PackedStringArray = []
	var feet := _player.global_position.y - 0.9
	var stack: Array = [_view]
	while not stack.is_empty():
		var n: Node = stack.pop_back()
		stack.append_array(n.get_children())
		if n is GeometryInstance3D and (n as GeometryInstance3D).is_visible_in_tree():
			var g := n as GeometryInstance3D
			var d := Vector2(g.global_position.x - _player.global_position.x, g.global_position.z - _player.global_position.z)
			if d.length() < 2.0:
				out.append("%s (%s) ← %s, h=%.2f" % [g.name, g.get_class(), g.get_parent().name, g.global_position.y - feet])
	mod.log_info("[pax_rpg] probe:\n" + "\n".join(out))
	return "%d объектов рядом — список в логе" % out.size()


## Боец арены (arena.gd): гоблин или орк-чемпион, как обычные враги локации.
func _arena_spawn(kind: String, pos: Vector3) -> Node:
	var g: Goblin = Orc.new() if kind == "orc" else Goblin.new()
	g.setup(mod, _player, _on_hero_hit, pos, _hero_attacking, _hero_blocking)
	if kind == "veteran":                                # ветеран: крепче, блок чаще, парирует удары героя
		g.veteran = true
		g.max_hp = 95.0
		g.hp = g.max_hp
		g.display_key = "pax_rpg_goblin_veteran"
		g.set("_guard_chance", 0.45)
	if g.has_signal("slammed"):
		g.connect("slammed", _orc_slam)
	_view.add_child(g)
	_goblins.append(g)
	return g


## Камера: свободная — сзади и сверху; при захвате врага плавно подлетает к герою и встаёт за правым плечом,
## глядя на середину между героем и врагом — обоих видно, удобно читать замахи.
## Кинокамера для превью (консоль F8: rpg_cam N, 0 — обычная): ракурсы «shots» из планировки города.
var _cine: Array = []
var _shots: Array = []


func cine(n: int) -> String:
	if n <= 0 or n > _shots.size():
		_cine = []
		if is_instance_valid(_hud):
			_hud.visible = true
		return "camera: player"
	_cine = _shots[n - 1]
	if is_instance_valid(_hud):
		_hud.visible = false
	return "camera: shot %d/%d" % [n, _shots.size()]


## Камера за спиной: на бегу вперёд и наискосок она плавно заходит за героя и смотрит по ходу движения. Вбок и на
## камеру — не крутится (иначе шаг вбок водил бы героя по кругу: движение считается от взгляда камеры). Крутил мышью —
## автоповорот ждёт camera_auto_wait с. camera_auto — скорость (0 — выкл.).
func _auto_yaw(delta: float) -> void:
	var rate := _f("camera_auto", 1.4)
	if rate <= 0.0 or not is_instance_valid(_player) or Input.mouse_mode != Input.MOUSE_MODE_CAPTURED:
		return
	if Time.get_ticks_msec() - _cam_mouse < int(_f("camera_auto_wait", 1.0) * 1000.0):
		return
	var v := Vector2(_player.velocity.x, _player.velocity.z)
	var sp := v.length()
	if sp < 0.5:
		return
	var off := angle_difference(_yaw, atan2(-v.x, -v.y))
	var w := maxf(0.0, cos(off))                     # вперёд — 1, наискосок — меньше, вбок и назад — 0
	var run := clampf(sp / maxf(_f("run_speed", 5.0), 0.1), 0.0, 1.0)
	_yaw += off * (1.0 - exp(-delta * rate * w * run))


func _follow_camera(delta := 0.0) -> void:
	if is_instance_valid(_iai) and _iai.cam_override:
		return                                       # камерой правит «Иай»
	if not _cine.is_empty() and is_instance_valid(_cam):
		var a: Array = _cine[0]
		var b: Array = _cine[1]
		_cam.global_position = Vector3(float(a[0]), float(a[1]), float(a[2]))
		_cam.look_at(Vector3(float(b[0]), float(b[1]), float(b[2])))
		return
	if not is_instance_valid(_cam):
		return
	var want := 1.0 if (_lock != null and is_instance_valid(_lock)) else 0.0
	_cam_lock = move_toward(_cam_lock, want, delta * _f("lock_camera_speed", 2.5)) if delta > 0.0 else want
	var k := smoothstep(0.0, 1.0, _cam_lock)
	var target := _player.position + Vector3(0.0, 0.9, 0.0)
	if delta > 0.0:                                  # камера догоняет героя с запаздыванием — не на жёсткой сцепке
		if _cam_follow == Vector3.INF or _cam_follow.distance_to(target) > 6.0:
			_cam_follow = target
		var kf := _hf("cam_follow", 9.0)
		_cam_follow = _cam_follow.lerp(target, 1.0 - exp(-delta * kf))
		_cam_follow.y = lerpf(_cam_follow.y, target.y, 1.0 - exp(-delta * kf * 1.6))
		target = _cam_follow
	# колесо: от обычной дали до «боевой» — как за плечом в захвате (высота и даль lock_camera_*)
	if delta > 0.0:
		_zoom = lerpf(_zoom, _zoom_want, 1.0 - exp(-delta * _f("zoom_smooth", 3.0)))
	var z := smoothstep(0.0, 1.0, _zoom)
	var ch := lerpf(_f("camera_height", 2.4), _f("camera_near_height", 1.6) - 0.9, z)
	var cd := lerpf(_f("camera_distance", 5.0), _f("camera_near_distance", 2.6), z)
	var free_pos := target + Vector3(0.0, ch, cd).rotated(Vector3.UP, _yaw)
	var look := target
	var pos := free_pos
	if k > 0.0:
		var back := Vector3(0.0, 0.0, 1.0).rotated(Vector3.UP, _yaw)
		var right := Vector3(1.0, 0.0, 0.0).rotated(Vector3.UP, _yaw)
		# за плечом (Z — сменить плечо): чуть сбоку и позади — враг открыт, спина не закрывает обзор
		var shoulder := _player.position + Vector3(0.0, _f("lock_camera_height", 1.7), 0.0) \
			+ right * _f("lock_camera_side", 1.05) * _shoulder + back * _f("lock_camera_distance", 3.2)
		var enemy := _lock_point if want == 0.0 else _lock.global_position + Vector3(0.0, 0.9, 0.0)
		if want > 0.0:
			_lock_point = enemy
		var lock_look := (_player.position + Vector3(0.0, 1.3, 0.0)).lerp(enemy, 0.65)
		pos = free_pos.lerp(shoulder, k)
		look = target.lerp(lock_look, k)
	if _shake > 0.0 and delta > 0.0:
		# тряска от попадания: короткая, затухает; смещаем и камеру, и точку взгляда — картинка вздрагивает целиком
		_shake = maxf(0.0, _shake - delta)
		var a := _shake_amp * _shake / maxf(_f("shake_time", 0.18), 0.01)
		var off := Vector3(randf_range(-1.0, 1.0), randf_range(-1.0, 1.0), randf_range(-1.0, 1.0)) * a
		pos += off
		look += off * 0.6
	if delta > 0.0 and is_instance_valid(_player):
		var dyn := _cam_dynamics(delta)
		pos += dyn
		look += dyn * 0.5
	var sway := _f("camera_sway", 0.0)                # по умолчанию выкл.: RPG, а не съёмка с плеча
	if sway > 0.0 and is_instance_valid(_player):
		# «ручная» камера (приём Bodycam): дыхание в покое, покачивание на бегу — кадр живой, а не на штативе
		var t := Time.get_ticks_msec() * 0.001
		var run := clampf(Vector2(_player.velocity.x, _player.velocity.z).length() / maxf(_f("run_speed", 5.0), 0.1), 0.0, 1.0)
		var amp := sway * (0.012 + 0.035 * run)
		var hand := Vector3(sin(t * 1.1) * 0.6 + sin(t * 2.3 + 1.7) * 0.4,
			sin(t * 1.7 + 0.5) * 0.5 + sin(t * 3.1) * 0.3 + sin(t * 9.0) * 0.25 * run, 0.0) * amp
		hand = _cam.global_transform.basis * hand
		pos += hand
		look += hand * 0.35
	if delta > 0.0 and _hero != null:
		# серия ударов: камера чуть ближе к герою (доля дистанции), после серии — обратно; игрок двинул мышь — гаснет
		var in_combo := bool(mod.get_setting("combat_cam", true)) and _hero.action in ["Attack1", "Attack2", "Attack3"]
		_combo_cam = move_toward(_combo_cam, 1.0 if in_combo else 0.0, delta / (0.35 if in_combo else 0.6))
		var e := smoothstep(0.0, 1.0, _combo_cam)
		pos = look + (pos - look) * (1.0 - _hf("cam_combo_push", 0.12) * e)
		_fov_punch = maxf(0.0, _fov_punch - delta / 0.28)
		var fov := _cam_fov0() - _hf("cam_final_fov", 5.0) * sin(_fov_punch * PI * 0.5)
		if absf(_cam.fov - fov) > 1e-6:                  # без толчка — тот же угол: камеру в сервер рендера не шлём
			_cam.fov = fov
	if _sult != null and _sult.active():             # кат-сцена ульты копья — своя камера
		var cc: Array = _sult.camera()
		pos = cc[0]
		look = cc[1]
	elif _blood != null and _blood.active():         # Кровожадная атака — своя камера
		var cb: Array = _blood.camera()
		pos = cb[0]
		look = cb[1]
	else:
		pos = _cam_collide(pos, delta)
	_cam.position = pos
	_cam.look_at(look, Vector3.UP.rotated((look - pos).normalized(), _cam_roll) if (look - pos).length() > 0.01 else Vector3.UP)

## Камера в движении: покачивание в такт шагам (вверх-вниз дважды за цикл шагов, вбок — раз), ходьба — легко,
## бег — сильнее; толчок от взмаха (вперёд и в сторону удара) и крен гаснут пружиной. config/location.json →
## cam_bob_walk / cam_bob_run (м), cam_step (м на шаг), cam_kick, cam_roll (рад); 0 — выкл.
## Камера не проходит сквозь стены: шар 0.3 м летит от головы героя к желаемому месту камеры; упёрся — камера
## встаёт перед препятствием. Подлетает к стене сразу, отходит обратно плавно (без рывков у углов).
var _cam_pull := 1.0                # доля пути от головы до камеры, что свободна (сглажена)
var _cam_shape: SphereShape3D
var _cam_q: PhysicsShapeQueryParameters3D
var _cam_q_rid := RID()


func _cam_collide(pos: Vector3, delta: float) -> Vector3:
	if not is_instance_valid(_player) or not _player.is_inside_tree() or not _f("cam_collide", 1.0) > 0.0:
		return pos
	var head := _player.global_position + Vector3(0, 0.9, 0)
	var motion := pos - head
	if motion.length() < 0.05:
		return pos
	if _cam_shape == null:
		_cam_shape = SphereShape3D.new()
		_cam_shape.radius = 0.3
	if _cam_q == null:                                   # один запрос на всё время, а не новый объект каждый такт
		_cam_q = PhysicsShapeQueryParameters3D.new()
		_cam_q.shape = _cam_shape
		_cam_q.collision_mask = 1
	var prid := _player.get_rid()
	if prid != _cam_q_rid:                               # новый герой (новая стройка) — исключаем его
		_cam_q_rid = prid
		_cam_q.exclude = [prid]
	var q := _cam_q
	q.transform = Transform3D(Basis(), head)
	q.motion = motion
	var r := _player.get_world_3d().direct_space_state.cast_motion(q)
	var free := clampf(float(r[0]) if r.size() > 0 else 1.0, 0.0, 1.0)
	if free < _cam_pull:
		_cam_pull = free                                     # к стене — сразу
	else:
		_cam_pull = lerpf(_cam_pull, free, 1.0 - exp(-delta * 4.0))   # от стены — плавно
	return head + motion * maxf(_cam_pull, 0.12)


func _cam_dynamics(delta: float) -> Vector3:
	var hs := Vector2(_player.velocity.x, _player.velocity.z).length()
	var on := _player.is_on_floor() and hs > 0.3
	var run := clampf((hs - _hf("walk_speed", 2.4)) / maxf(_hf("run_speed", 5.0) - _hf("walk_speed", 2.4), 0.1), 0.0, 1.0)
	var want := (lerpf(_f("cam_bob_walk", 0.01), _f("cam_bob_run", 0.025), run) if on else 0.0)
	_bob_amp = lerpf(_bob_amp, want, 1.0 - exp(-delta * 6.0))
	_bob_ph += hs * delta / maxf(_f("cam_step", 0.85) * (1.0 + run * 0.35), 0.1) * PI   # пол-оборота — шаг
	var v := absf(sin(_bob_ph)) * _bob_amp                      # вниз на каждом шаге (период |sin| — шаг)
	var h := sin(_bob_ph) * _bob_amp * 0.4                      # вбок — с ноги на ногу
	# пружина толчка (Гук с демпфером): x'' = −k·x − c·x'
	_cam_kick_v += (-_cam_kick * _f("cam_kick_k", 140.0) - _cam_kick_v * _f("cam_kick_damp", 16.0)) * delta
	_cam_kick += _cam_kick_v * delta
	_cam_roll = lerpf(_cam_roll, sin(_bob_ph) * _bob_amp * _f("cam_bob_roll", 0.0), 1.0 - exp(-delta * 7.0))   # крен от шагов — выкл.: укачивает
	var b := _cam.global_transform.basis
	return b * Vector3(h + _cam_kick.x, _bob_amp * 0.5 - v + _cam_kick.y, _cam_kick.z)


## Толчок камеры на взмахе: к герою (вперёд) и в сторону удара, крен по взмаху; финальный — сильнее.
func _cam_swing(move: String) -> void:
	var k := _f("cam_kick", 0.05) * (1.8 if move == "Attack3" else 1.0)
	var side := 0.0
	if _hero != null:
		side = float(_hero.swing_side) if _hero.swing_side != 0 else (-1.0 if move == "Attack1" else 1.0)
	_cam_kick_v += Vector3(side * k * 4.0, -k * 1.5, -k * 9.0)       # z < 0 — к герою
	_cam_roll += side * _f("cam_roll", 0.008) * (1.6 if move == "Attack3" else 1.0)

func _impact_feel(gob: Goblin, heavy: bool, heft := 1.0) -> void:
	_sfx_once("sounds/impact_%d.wav" % randi_range(1, 3), -7.0 + 3.0 * clampf(heft - 1.0, -1.0, 1.0))   # меч попал
	var t := (_hf("hitstop_heavy", 0.14) if heavy else _hf("hitstop", 0.08)) * clampf(heft, 0.6, 1.6)   # F = Δp/Δt
	if _hero != null:
		_hero.hitstop(t)
	if is_instance_valid(gob):
		gob.hitstop(t)
	_shake = _f("shake_time", 0.18)
	_cam_kick_v += Vector3(0.0, -0.8, -1.2) * clampf(heft, 0.6, 1.8) * _f("cam_hit_kick", 1.0)   # вздрог к удару
	_shake_amp = (_hf("shake_heavy", 0.09) if heavy else _hf("shake", 0.045)) * clampf(heft, 0.6, 1.8)

## Алый блик по краям экрана (финал серии): над 3D, под интерфейсом; гаснет за fx_flash_time.
func _flash_tick(delta: float) -> void:
	if _flash_t <= 0.0 and not is_instance_valid(_cflash):
		return
	if not is_instance_valid(_cflash):
		_cflash = ColorRect.new()
		_cflash.name = "PaxRpgComboFlash"
		_cflash.set_anchors_preset(Control.PRESET_FULL_RECT)
		_cflash.mouse_filter = Control.MOUSE_FILTER_IGNORE
		var sh := Shader.new()
		sh.code = FLASH
		var m := ShaderMaterial.new()
		m.shader = sh
		_cflash.material = m
		add_child(_cflash)
		move_child(_cflash, _box.get_index() + 1 if is_instance_valid(_box) else 0)
	_flash_t = maxf(0.0, _flash_t - delta)
	var a := _flash_t / maxf(_hf("fx_flash_time", 0.22), 0.01)
	var col: Array = hero_cfg.get("mana_color", [1.0, 0.08, 0.16])
	(_cflash.material as ShaderMaterial).set_shader_parameter("tint", Color(float(col[0]), float(col[1]), float(col[2]), a * a))
	_cflash.visible = a > 0.001


const FLASH := """shader_type canvas_item;
uniform vec4 tint = vec4(1.0, 0.1, 0.2, 0.0);
void fragment() {
	vec2 d = UV - vec2(0.5);
	float v = smoothstep(0.3, 0.75, length(d * vec2(1.0, 0.8)));

}"""


## Удар героя по врагам рядом: в пределах range и перед ним (arc — косинус полуугла; -1 — по кругу, удар о землю).
func _strike(hit: Dictionary) -> void:
	var fwd := -_body_mesh.global_transform.basis.z
	fwd.y = 0.0
	fwd = fwd.normalized()
	for g: Variant in _goblins:
		if not is_instance_valid(g) or not (g as Goblin).is_alive():
			continue
		var gob: Goblin = g
		var to := gob.global_position - _player.global_position
		to.y = 0.0
		var d := to.length()
		if d > float(hit.range) + 0.35:
			continue
		if d > 0.3 and fwd.dot(to / d) < float(hit.arc):
			continue
		var res := gob.take_damage(float(hit.damage) * _dmg_mult(), _player.global_position, bool(hit.get("knock", false)))
		if _strike_result(gob, res, float(hit.damage)):
			continue
		if _hero != null and _hero.action == "Launcher":
			_sfx_once("sounds/kick_hit.wav", 0.0)          # пинок попал
		if _hero != null and _hero.action == "Launcher" and gob.is_alive():   # подброс: пинок по площади — враг взлетает
			gob.launch(Vector3(0.0, 8.5 if not gob.boss else 3.0, 0.0))
		_style.hit(_hero.action if _hero != null and _hero.action != "" else "Attack3")
		_impact_feel(gob, bool(hit.get("knock", false)), _mass_hit(gob, 0.0, bool(hit.get("knock", false))))
		_numbers.append({"pos": gob.global_position + Vector3(randf_range(-0.3, 0.3), 1.3, 0.0),
			"v": int(hit.damage), "crit": bool(hit.get("knock", false)), "t": 0.0})
		_combat_fx_hit(gob, hit)
		if bool(hit.get("final", false)) and gob.is_alive() and not gob.boss:   # финал серии — алой энергией отбрасывает
			var away := to / d if d > 0.3 else fwd
			gob.launch(away * _hf("final_push", 6.5) + Vector3(0.0, _hf("final_push_up", 2.2), 0.0))


## Масса в ударе: импульс p = m·v (острие v = ω·r), враг отлетает на Δv = p/m_врага (сохранение импульса), герою —

			add_child(_voice)

	var asked := mod.ask_ai("foe_line", prompt, {}, got, {"temperature": 0.9, "max_tokens": 60})
	if not asked:
		_foe_line_wait = false


func _mass_hit(gob: Goblin, tip: float, heavy: bool) -> float:
	# острие: прошлое положение в первом кадре взмаха бывает старым — скорость не больше 2.2 × ω·r
	if gob.boss and gob.is_alive():
		_foe_line(gob, "got_hit")
	var p := _w.impulse(minf(tip, _w.f("weapon_omega", 9.0) * _w.f("weapon_length", 1.3) * 2.2), heavy)
	var me := gob.mass * (_w.f("boss_mass_mult", 2.2) if gob.boss and gob.mass < 60.0 else 1.0)   # главарь гоблинов — тяжелее
	var away := gob.global_position - _player.global_position
	away.y = 0.0
	away = away.normalized() if away.length() > 0.01 else -_body_mesh.global_transform.basis.z
	if gob.is_alive():
		var kv := _w.knock(p, me)
		var flat := Vector2(gob.velocity.x, gob.velocity.z)
		if heavy:                                           # финальный удар: отлёт тоже по массе (орка — меньше)
			var ref := _w.knock(_w.impulse(0.0, true), _w.f("goblin_mass", 45.0))
			flat *= clampf(kv / maxf(ref, 0.01), 0.25, 1.4)
		else:
			flat = Vector2(away.x, away.z) * kv
		gob.velocity.x = flat.x
		gob.velocity.z = flat.y
	_recoil -= away * minf(_w.recoil(p, me), _w.f("recoil_max", 0.9))     # толчок, а не полёт
	if _recoil.length() > _w.f("recoil_max", 0.9) * 1.5:                    # серия ударов не складывается в разгон
		_recoil = _recoil.normalized() * _w.f("recoil_max", 0.9) * 1.5
	return _w.heft(p) * (1.25 if heavy else 1.0)


## Удар клинком по пути: враг задет, если полоса клинка (сейчас, кадр назад и посередине) прошла ближе
## blade_radius к его телу (вертикальный отрезок от колен до головы). Один взмах — один урон каждому.
var _blade_hit := {}                 # id взмаха → {враг: true}


func _blade_strike(b: Dictionary) -> void:
	var id := int(b.id)
	if not _blade_hit.has(id):
		_blade_hit = {id: {}}
	var done: Dictionary = _blade_hit[id]
	# полоса клинка между прошлым и нынешним кадром — с шагом ~blade_sub_step по острию: тяжёлый меч в ударе
	# пролетает за кадр физики до метра, и три положения могли проскочить врага
	var segs := []
	var tip_step := ((b.b as Vector3) - (b.pb as Vector3)).length()
	var n := clampi(int(ceil(tip_step / _hf("blade_sub_step", 0.25))), 2, 12)
	for i in n + 1:
		var k := float(i) / float(n)
		segs.append([(b.pa as Vector3).lerp(b.a as Vector3, k), (b.pb as Vector3).lerp(b.b as Vector3, k)])
	var rad := _hf("blade_radius", 0.55)
	for g: Variant in _goblins:
		if not is_instance_valid(g) or not (g as Goblin).is_alive() or done.has(g):
			continue
		var gob: Goblin = g
		var lo := gob.global_position + Vector3(0, 0.3, 0)
		var hi := gob.global_position + Vector3(0, 1.6 if gob.boss else 1.2, 0)
		var hit := false
		for sg: Array in segs:
			var cp := Geometry3D.get_closest_points_between_segments(sg[0], sg[1], lo, hi)
			if (cp[0] as Vector3).distance_to(cp[1]) < rad + (0.35 if gob.boss else 0.0):
				hit = true
				break
		if not hit:
			continue
		done[g] = true
		var dmg := float(b.damage) * _dmg_mult()
		var res := gob.take_damage(dmg, _player.global_position, false)
		if _strike_result(gob, res, float(b.damage)):
			continue
		var mv := str(b.get("move", "Attack1"))
		_style.hit(mv)

		var tip := ((b.b as Vector3) - (b.pb as Vector3)).length() / maxf(get_physics_process_delta_time(), 0.001)
		_impact_feel(gob, false, _mass_hit(gob, tip, false))
		_numbers.append({"pos": gob.global_position + Vector3(randf_range(-0.3, 0.3), 1.3, 0.0), "v": int(dmg),
			"crit": false, "t": 0.0})


## Ответ врага на удар героя: «guard» — принял на клинок (атакующему дороже: герой тратит больше, ДИЗДОК — бой),
## «parry» — враг отбил: концентрация героя — 0 (слом). true — удар не прошёл (без цифр и крови).
func _strike_result(gob: Goblin, res: String, base_dmg: float) -> bool:
	match res:
		"guard":
			_spend(base_dmg * _hf("conc_hit_blocked", 0.5) * (1.5 if _conc < _conc_max() / 3.0 else 1.0), gob)
			_sfx_once("sounds/katana_hit_%d.ogg" % randi_range(0, 3), -5.0)
			_sparks(gob.global_position + Vector3(0.0, 1.0, 0.0), Color(1.0, 0.65, 0.3), 16, 3.5)
			if _hero != null:
				_hero.hitstop(0.05)
			return true
		"parry":
			_parry_sound()
			_sparks(gob.global_position + Vector3(0.0, 1.1, 0.0), Color(1.0, 0.85, 0.4), 44, 5.5)
			_shake = 0.25
			_shake_amp = 0.06
			_hero_break(gob)
			return true
		"broken":
			_toast = {"text": mod.tr_key("pax_rpg_conc_enemy_broken"), "t": 0.0}
	return false

## Куда бить: в захваченного врага; без захвата — в сторону движения (WASD от камеры); без ввода — INF (как стоит).
func _combo_yaw() -> float:
	if _lock != null and is_instance_valid(_lock):
		return _aim_yaw()
	if _typing():
		return INF
	var x := float(Input.is_physical_key_pressed(KEY_D)) - float(Input.is_physical_key_pressed(KEY_A))
	var y := float(Input.is_physical_key_pressed(KEY_S)) - float(Input.is_physical_key_pressed(KEY_W))
	var base := _body_mesh.rotation.y                     # без ввода — куда стоит
	if absf(x) + absf(y) >= 0.01:
		var m := Vector3(1.0, 0.0, 0.0).rotated(Vector3.UP, _yaw) * x - Vector3(0.0, 0.0, -1.0).rotated(Vector3.UP, _yaw) * y
		base = atan2(-m.x, -m.z)
	# магнит: ближайший живой враг в конусе этого направления — удар в него (каждый удар серии заново)
	var best: Goblin = null
	var bd := _hf("magnet_range", 5.0)
	var cone := deg_to_rad(_hf("magnet_angle", 75.0))
	for g: Variant in _goblins:
		if not is_instance_valid(g) or not (g as Goblin).is_alive():
			continue
		var gob: Goblin = g
		var d := gob.global_position.distance_to(_player.global_position)
		if d < bd and absf(angle_difference(base, _yaw_to(gob))) < cone:
			bd = d
			best = gob
	if best != null:
		return _yaw_to(best)
	return base if absf(x) + absf(y) >= 0.01 else INF


func _aim_yaw() -> float:
	if _lock != null:
		var d := _lock.global_position - _player.global_position
		return atan2(-d.x, -d.z)
	return _yaw


## Магнит удара: герой доворачивается к врагу, но несёт его только шаг клипа — не дотягивает к врагу (было:
## «телепорт» вплотную; вернуть — magnet_pull: true в hero_*.json). Ближе reach шаг клипа не несёт — не влезает во врага.
func _magnet_pull(vel: Vector3, mag: Goblin, delta: float) -> Vector3:
	var to := mag.global_position - _player.global_position
	to.y = 0.0
	var dist := to.length()
	if dist < 0.01 or delta <= 0.0:
		return vel
	var dir := to / dist
	var reach := _hero.attack_reach()
	var gap := dist - reach
	var along := vel.x * dir.x + vel.z * dir.z
	var tl := _hero.hit_time_left()
	if bool(hero_cfg.get("magnet_pull", false)) and tl >= 0.0 and _hero.pull_on() and gap > 0.0:   # дотяг — по желанию
		var need := minf(gap / maxf(tl, 0.06), _hf("magnet_pull_max", 8.0))
		if need > along:
			vel.x += dir.x * (need - along)
			vel.z += dir.z * (need - along)
			along = need
	if along > 0.0 and along * delta > gap:                  # за кадр не проскочить ближе reach
		var cut := along - maxf(gap, 0.0) / delta
		vel.x -= dir.x * cut
		vel.z -= dir.z * cut
	return vel


## Цель удара: захваченный враг, иначе ближайший живой в magnet_range перед камерой (в пределах magnet_angle),
## а вплотную (magnet_close) — с любой стороны.
func _attack_target() -> Goblin:
	if _lock != null and is_instance_valid(_lock) and (_lock as Goblin).is_alive():
		return _lock as Goblin
	var best: Goblin = null
	var bd := _hf("magnet_range", 5.0)
	var cone := deg_to_rad(_hf("magnet_angle", 75.0))
	var ref := _body_mesh.rotation.y if _hero != null and _hero.action in Hero.ATTACKS else _yaw
	for g: Variant in _goblins:
		if not is_instance_valid(g):
			continue
		var gob: Goblin = g
		if not gob.is_alive():
			continue
		var d := gob.global_position.distance_to(_player.global_position)
		if d < bd and (d < _hf("magnet_close", 3.0) or absf(angle_difference(ref, _yaw_to(gob))) < cone):
			bd = d
			best = gob
	return best


func _yaw_to(n: Node3D) -> float:
	var d := n.global_position - _player.global_position
	return atan2(-d.x, -d.z)


func _face_now(yaw: float) -> void:
	_body_mesh.rotation.y = yaw

## Q: захватить ближнего врага перед камерой (до 16 м) / отпустить.
func _toggle_lock() -> void:
	if _lock != null:
		_set_lock(null)
		return
	var cam_f := Vector3(-sin(_yaw), 0.0, -cos(_yaw))
	var best: Node3D = null
	var best_s := INF
	for g: Variant in _goblins:
		if not is_instance_valid(g) or not (g as Goblin).is_alive():
			continue
		var gob: Goblin = g
		var to := gob.global_position - _player.global_position
		to.y = 0.0
		var d := to.length()
		if d > 16.0:
			continue
		var score := d * (2.0 - cam_f.dot(to / maxf(d, 0.01)))   # ближе и прямее перед камерой — лучше
		if score < best_s:
			best_s = score
			best = gob
	_set_lock(best)


## В захвате мышь не крутит камеру — рывок мыши в сторону переводит захват на врага, что на экране в той стороне
## (ближайший по горизонтали к прицелу). Жест — сумма движения за короткое время (пауза — заново).
var _flick := Vector2.ZERO
var _flick_ms := 0
var _flick_cd_ms := 0


func _lock_flick(rel: Vector2) -> void:
	var now := Time.get_ticks_msec()
	if now - _flick_ms > 220:
		_flick = Vector2.ZERO
	_flick_ms = now
	_flick += rel * _sens
	if now < _flick_cd_ms or absf(_flick.x) < _hf("lock_flick_px", 90.0) or not is_instance_valid(_cam) \
			or not is_instance_valid(_lock):
		return
	var cur := _cam.unproject_position(_lock.global_position + Vector3(0.0, 0.9, 0.0))
	var sgn := signf(_flick.x)
	var best: Goblin = null
	var best_s := INF
	for g: Variant in _goblins:
		if not is_instance_valid(g) or g == _lock or not (g as Goblin).is_alive():
			continue
		var gob: Goblin = g
		var p := gob.global_position + Vector3(0.0, 0.9, 0.0)
		if gob.global_position.distance_to(_player.global_position) > 22.0 or _cam.is_position_behind(p):
			continue
		var sp := _cam.unproject_position(p)
		var dx := (sp.x - cur.x) * sgn
		if dx < 24.0:
			continue
		var score := dx + absf(sp.y - cur.y) * 1.5
		if score < best_s:
			best_s = score
			best = gob
	_flick = Vector2.ZERO
	if best != null:
		_set_lock(best)
		_flick_cd_ms = now + 350
		_sfx_once("sounds/air_whoosh.wav", -16.0)


func _set_lock(t: Node3D) -> void:
	_lock = t
	if is_instance_valid(_lock_mark):
		_lock_mark.visible = false


## ЛКМ рядом со сломленным врагом (концентрация 0, на коленях) — Кровожадная атака вместо удара: кат-сцена blood.gd.
## Захваченный — первым, иначе ближайший в blood_range м.
var _blood: Blood


func _try_bloodthirsty() -> bool:
	if _hero == null or (_blood != null and _blood.active()):
		return false
	var best: Goblin = null
	var bd := _hf("blood_range", 3.2)
	for g: Variant in _goblins:
		if not is_instance_valid(g):
			continue
		var gob: Goblin = g
		if not gob.can_deathblow():
			continue
		var to := gob.global_position - _player.global_position
		to.y = 0.0
		var d := to.length()

		if d < bd:

	if best == null:
		return false
	var dir := best.global_position - _player.global_position
	_face_now(atan2(-dir.x, -dir.z))
	var hit_at := _hero.bloodthirsty()
	if hit_at < 0.0:
		return false
	if _blood == null:
		_blood = Blood.new()
		add_child(_blood)
	_blood.start(self, _hero, best, hit_at)
	return true


## Удар врага по герою: кувырок — мимо; парирование в первые доли секунды блока — отбит, концентрация врага — 0
## (в сериях босса — часть); блок — урона нет, концентрация тратится у обоих (у атакующего больше); иначе — урон
## и трата концентрации. Герой сломлен (концентрация 0) — удар врага становится Кровожадной атакой: треть здоровья.
## kind: "heavy" — блок тратит больше; "perilous" — не блокируется и не парируется (только уйти);
## "charge" — таран: блок не держит, сбивает с ног; "parried" — враг отбил удар героя: слом.
func _on_hero_hit(damage: float, attacker: Node3D = null, kind := "") -> void:
	if _hero != null and _hero.invulnerable():
		return
	if _blood != null and _blood.active():
		return
	if damage > 0.0 and attacker is Goblin and (attacker as Goblin).boss:
		_foe_line(attacker as Goblin, "hit_hero")
	if is_instance_valid(_iai) and _iai.active and not _iai.interrupt():
		return                                       # время стоит — по герою не попасть
	if kind == "parried":
		_hero_break(attacker)
		_update_hud()
		return
	if _broken_t > 0.0 and damage > 0.0:            # сломлен: Кровожадная атака врага — треть здоровья, сбит с ног
		_hp = maxf(0.0, _hp - maxf(damage, _hp_max() / 3.0))
		_broken_t = 0.0
		_conc = _conc_max() * _hf("conc_after_execute", 0.5)
		_style.hurt()
		_armor_hit_sound(40.0)
		_sfx_once("sounds/deathblow.wav", -2.0)
		_shake = 0.45
		_shake_amp = 0.11
		_flash_t = _hf("fx_flash_time", 0.22) * 2.0
		_sparks(_player.global_position + Vector3(0.0, 1.1, 0.0), Color(0.6, 0.02, 0.05), 50, 5.0)
		_toast = {"text": mod.tr_key("pax_rpg_hero_executed"), "t": 0.0}
		if _hp > 0.0 and _hero != null:
			_hero.knock_down()
			if attacker != null:
				var push0 := _player.global_position - attacker.global_position
				push0.y = 0.0
				_shove = push0.normalized() * _hf("execute_shove", 5.0)
		_hero_down_check(attacker)
		return
	var unblockable := kind == "perilous" or kind == "charge"
	var fwd := Vector3(-sin(_body_mesh.rotation.y), 0.0, -cos(_body_mesh.rotation.y))
	var facing := true
	if attacker != null:
		var to := attacker.global_position - _player.global_position
		to.y = 0.0
		facing = to.length() < 0.01 or fwd.dot(to.normalized()) > 0.2
	var spark_at := _player.global_position + fwd * 0.7 + Vector3(0.0, 1.2, 0.0)
	var heavy := kind == "heavy"
	if _hero != null and facing and not unblockable and _hero.parry_window():
		_hero.parry_anim()                           # отбил: клип Parry, блок остаётся
		_style.hit("Parry")
		_parry_sound(attacker != null and str(attacker.get("_atk")) == "Kick")   # пинок — глухой шлепок, не звон стали
		_sparks(spark_at, Color(1.0, 0.8, 0.35), 40, 5.0)
		_shake = maxf(_shake, 0.12)
		_shake_amp = 0.04
		if attacker is Goblin:
			(attacker as Goblin).deflected(_player.global_position)   # концентрация врага — 0 (в серии босса — часть)
			if (attacker as Goblin).can_deathblow():
				_toast = {"text": mod.tr_key("pax_rpg_conc_enemy_broken"), "t": 0.0}
		return
	if _hero != null and facing and not unblockable and _hero.blocking:
		var guard := 1.0 - _block_reduce()               # пассивка героя (Чёрный рыцарь — «Железная стена»)
		var tired := 1.5 if _conc < _conc_max() / 3.0 else 1.0   # устал — блок держит хуже
		_since_hit = 0.0
		_sparks(spark_at, Color(1.0, 0.6, 0.25), 14, 3.0)
		_hero.block_hit(heavy)                          # отдача: сдвиг назад — root motion клипа
		if attacker is Goblin:
			(attacker as Goblin).on_blocked(damage, heavy)   # атакующий тратит больше блокирующего
		_spend(damage * (0.75 if heavy else 0.45) * tired * guard, attacker)
		return
	_hp = maxf(0.0, _hp - damage)
	_style.hurt()
	if attacker != null and str(attacker.get("_atk")) == "Kick":
		_sfx_once("sounds/kick_hit.wav", 0.0)              # пинок врага — удар в живот, а не по стали
	else:
		_armor_hit_sound(damage)
	_since_hit = 0.0
	if _hero != null:
		if kind == "charge" and _hp > 0.0:
			_hero.knock_down()
			if attacker != null:                             # таран: героя отбрасывает и он скользит по земле
				var push := _player.global_position - attacker.global_position
				push.y = 0.0
				_shove = push.normalized() * _hf("charge_shove", 9.0)
				_player.velocity.y = _hf("charge_lift", 3.5)
		else:
			_hero.hurt()
	if _hp > 0.0:
		_spend(clampf(damage * _hf("conc_hit_taken", 0.7), 6.0, 30.0), attacker)
	_hero_down_check(attacker)


## Здоровье кончилось: на арене — уносят с песка (без потери прогресса), в городе — на старт.
func _hero_down_check(attacker: Node3D) -> void:
	if _hp <= 0.0 and attacker != null and is_instance_valid(attacker) and attacker.has_method("cheer"):
		attacker.call("cheer")                       # орк ликует над павшим героем
	if _hp <= 0.0 and is_instance_valid(_arena) and _arena.fighting:
		_hp = 1.0
		_broken_t = 0.0
		_arena.leave(true)
	if _hp <= 0.0 and is_instance_valid(_player):
		_hp = _hp_max()
		_hp_ghost = 1.0
		_conc = _conc_max()
		_broken_t = 0.0
		_flask = int(_hf("flask_charges", 3.0))
		_player.position = START
		_player.velocity = Vector3.ZERO
	_update_hud()

	if _db_snd == null:

func _regen_conc(delta: float) -> void:
	_conc_idle += delta
	if _broken_t > 0.0 or _conc >= _conc_max() or _conc_idle < CONC_IDLE:
		return
	if _hero != null and (_hero.blocking or _hero.action in Hero.ATTACKS or _hero.action in Hero.SPECIALS):
		return
	var near := _foe_near(_hf("conc_near", 4.5))
	var k := (1.0 if near else _hf("conc_far_mult", 1.6)) * (0.35 + 0.65 * _hp / maxf(_hp_max(), 1.0))
	_conc = minf(_conc_max(), _conc + _hf("conc_regen", CONC_REGEN) * k * delta)

## LUT 17³ (цветокоррекция сцены): тени уходят в холодную бирюзу, света — в тёплое золото, мягкая S-кривая.
## Делаем в коде — без файлов и без потерь при сжатии.
static var _lut: ImageTexture3D


static func _grade_lut() -> ImageTexture3D:
	if _lut != null:
		return _lut                                  # одна на сессию (раньше новая текстура на каждый вход в город)
	var n := 17
	var layers: Array[Image] = []
	var lilac := Vector3(0.9, 0.97, 1.08)            # тени — в холодную бирюзу поверхности Юрумы (контраст тёплым огням)
	var warm := Vector3(1.07, 1.0, 0.88)
	for b: int in n:
		var img := Image.create(n, n, false, Image.FORMAT_RGB8)
		for g: int in n:
			for r: int in n:
				var c := Vector3(r, g, b) / float(n - 1)
				var lum := c.dot(Vector3(0.2126, 0.7152, 0.0722))
				var ws := 1.0 - smoothstep(0.0, 0.45, lum)
				var wh := smoothstep(0.45, 1.0, lum)
				c = c.lerp(c * lilac + Vector3(0.0, 0.012, 0.03), ws * 0.65)
				c = c.lerp(c * warm, wh * 0.55)
				c += (c - Vector3.ONE * 0.5) * 0.06
				img.set_pixel(r, g, Color(clampf(c.x, 0, 1), clampf(c.y, 0, 1), clampf(c.z, 0, 1)))
		layers.append(img)
	_lut = ImageTexture3D.new()
	_lut.create(Image.FORMAT_RGB8, n, n, n, false, layers)
	return _lut

func _tick_orbs(delta: float) -> void:
	_style.tick(delta)
	_lvl_flash = maxf(0.0, _lvl_flash - delta)
	_gold_flash = maxf(0.0, _gold_flash - delta * 0.8)

		return
	var keep: Array = []
	for o: Dictionary in _orbs:

		if float(o.t) - float(o.delay) >= ORB_TIME:
			var ups := _prog.add_xp(float(o.v))
			if ups > 0:
				_lvl_flash = 1.6

					_toast = {"text": mod.tr_key("pax_rpg_hud_skill_point") % _prog.level, "t": 0.0}
		else:
			keep.append(o)
	_orbs = keep


const ORB_TIME := 0.95


func _orb_points() -> Array:
	var out: Array = []
	if not is_instance_valid(_hud):
		return out
	var target := _hud.xp_target(_prog.frac())
	for o: Dictionary in _orbs:
		var k := clampf((float(o.t) - float(o.delay)) / ORB_TIME, 0.0, 1.0)
		if k <= 0.0:
			continue
		var p0: Vector2 = o.p0
		var ctrl := p0.lerp(target, 0.4) + Vector2(-160.0, -220.0) * (0.6 + 0.4 * float(o.bend))
		var trail: Array = []
		for j: int in range(5, 0, -1):                     # шлейф: где шарик был чуть раньше
			trail.append(_bez(p0, ctrl, target, maxf(0.0, k - j * 0.035)))
		out.append([_bez(p0, ctrl, target, k), k, trail])
	return out


static func _bez(p0: Vector2, c: Vector2, p1: Vector2, k: float) -> Vector2:
	var e := k * k * (3.0 - 2.0 * k)
	return p0.lerp(c, e).lerp(c.lerp(p1, e), e)

func _hud_tick(delta: float) -> void:

			or not _player.is_inside_tree():

			_dead_seen[id] = true
			_kills += 1
			# мана не растворяется: шарики перелетают в шкалу опыта (левый верхний угол), опыт — по прилёту
			var gain := float((hero_cfg.get("progress", {}) as Dictionary).get("xp_per_kill", 35.0))

				_t("pax_rpg_gold_got") % coins, "t": 0.0}
		elif gob.is_alive():
			_dead_seen[id] = false
	_tick_orbs(delta)
	var hp := _hp / _hp_max()
	if hp >= _hp_ghost:
		_hp_ghost = hp
	elif _since_hit > 0.6:
		_hp_ghost = move_toward(_hp_ghost, hp, delta * 0.6)
	_hint_t = maxf(0.0, _hint_t - delta)

		"skill_points": _prog.skill_points, "gold": _prog.gold, "gold_flash": _gold_flash,
		"style": {"letter": _style.letter(), "rank": _style.rank(), "frac": _style.frac(), "flash": _style.flash,
			"title": _t("pax_rpg_style_" + _style.letter().to_lower()) if _style.letter() != "" else ""}, "orbs": _orb_points(), "lvl_flash": _lvl_flash, "flask": _flask, "flask_max": int(_hf("flask_charges", 3.0)),
		"yaw": _yaw, "body_yaw": _body_mesh.rotation.y if _body_mesh != null else _yaw, "kills": _kills,
		"me": Vector2(_player.global_position.x, _player.global_position.z), "map_rects": _map_rects,
		"gate": Vector2(_gate_pos.x, _gate_pos.z), "near_gate": _near_gate, "place": _city_name,
		"hint": _t("pax_rpg_controls"), "hint_a": clampf(_hint_t, 0.0, 1.0),
		"gate_hint": _t("pax_rpg_gate_hint") if _near_gate else ""}
	var targets: Array = []
	var enemies: Array = []
	for g: Variant in _goblins:
		if not is_instance_valid(g) or not (g as Goblin).is_alive():
			continue
		var gob: Goblin = g
		var d := gob.global_position.distance_to(_player.global_position)
		var locked := gob == _lock
		enemies.append({"pos": Vector2(gob.global_position.x, gob.global_position.z), "locked": locked})
		if d < 45.0:
			targets.append({"yaw": _yaw_to(gob), "dist": d, "color": Color(1.0, 0.36, 0.56) if locked
				else Color(1.0, 0.55, 0.78)})
	targets.append({"yaw": atan2(-(_gate_pos.x - _player.global_position.x), -(_gate_pos.z - _player.global_position.z)),
		"dist": Vector2(_gate_pos.x - _player.global_position.x, _gate_pos.z - _player.global_position.z).length(),
		"color": Color(0.47, 0.84, 1.0), "ring": true})
	st["targets"] = targets

	if e != null:
		st["enemy"] = {"name": _t(e.display_key), "hp": e.hp / e.max_hp,
			"conc": e.conc / e.conc_max, "broken": e.can_deathblow(), "boss": e.boss, "phase": e.phase,
			"veteran": e.veteran, "id": e.get_instance_id(),
			"title": _t(e.title_key) if e.title_key != "" else ""}
		st["ult"] = 1.0 - e.conc / e.conc_max if not e.can_deathblow() else 1.0

	if _lock != null and is_instance_valid(_lock):
		var lp := _lock.global_position + Vector3(0.0, 0.85, 0.0)
		if not _cam.is_position_behind(lp):
			st["lock_screen"] = _cam.unproject_position(lp)
			st["lock_broken"] = (_lock as Goblin).can_deathblow()
	var nums: Array = []
	for i: int in range(_numbers.size() - 1, -1, -1):
		var n: Dictionary = _numbers[i]
		n.t = float(n.t) + delta
		if float(n.t) > 1.1:
			_numbers.remove_at(i)
			continue
		var wp: Vector3 = n.pos + Vector3(0.0, float(n.t) * 0.7, 0.0)

			continue

	if not _toast.is_empty():

		else:
			st["toast"] = {"text": _toast.text, "a": 1.0 - smoothstep(2.8, 3.5, float(_toast.t))}
	var act := _hero.action if _hero != null else ""

const KEYS := [KEY_W, KEY_A, KEY_S, KEY_D, KEY_E, KEY_F, KEY_Q, KEY_SPACE, KEY_SHIFT, KEY_ALT, KEY_C, KEY_R, KEY_H, KEY_M, KEY_X, KEY_T, KEY_V,
	KEY_Z, KEY_TAB, KEY_J, KEY_G]


func _input(event: InputEvent) -> void:
	if not active:
		return
	# клавиши скорости игры (1–3, P, пробел) игра слушает раньше нас и включает «наблюдение» — оно мотает мир;
	# сразу после её обработчика возвращаем паузу (пробел — прыжок, его держит _hold_pause)
	if event is InputEventKey and (event as InputEventKey).pressed and not (event as InputEventKey).echo:
		var sk := (event as InputEventKey).physical_keycode
		if sk in [KEY_1, KEY_2, KEY_3, KEY_P, KEY_SPACE]:
			_stop_world.call_deferred()
	if paused():
		if event is InputEventKey and (event as InputEventKey).pressed and not (event as InputEventKey).echo:
			var pk := (event as InputEventKey).physical_keycode
			if is_instance_valid(_journal) and _journal.visible and (pk == KEY_ESCAPE or pk == KEY_J):
				_set_journal(false)
				get_viewport().set_input_as_handled()
			elif is_instance_valid(_city_map) and _city_map.visible and (pk == KEY_ESCAPE or pk == KEY_M):
				_set_city_map(false)
				get_viewport().set_input_as_handled()
			elif is_instance_valid(_talk) and _talk.visible and pk == KEY_ESCAPE:
				_talk.close()
				get_viewport().set_input_as_handled()
			elif is_instance_valid(_wrist) and _wrist.visible and pk == KEY_ESCAPE:
				_set_wrist(false)
				get_viewport().set_input_as_handled()
			elif is_instance_valid(_inv) and _inv.visible and pk == KEY_ESCAPE:
				if not _inv.back():                       # сначала — окно оружия, потом сам инвентарь
					_set_inventory(false)
				get_viewport().set_input_as_handled()
			elif pk == KEY_ESCAPE:
				_set_paused(false)
				get_viewport().set_input_as_handled()
		return
	if event is InputEventMouseButton:
		var mb: InputEventMouseButton = event
		if mb.pressed and mb.button_index in [MOUSE_BUTTON_WHEEL_UP, MOUSE_BUTTON_WHEEL_DOWN]:
			_zoom_want = clampf(_zoom_want + (0.1 if mb.button_index == MOUSE_BUTTON_WHEEL_UP else -0.1), 0.0, 1.0)
			get_viewport().set_input_as_handled()
			return
		if Input.mouse_mode != Input.MOUSE_MODE_CAPTURED:
			if is_instance_valid(_help) and (_help.is_open() or _help.over(mb.position)):
				return                                   # клик — по кнопке / окну «Управление»
			if mb.pressed and mb.button_index == MOUSE_BUTTON_LEFT:
				Input.mouse_mode = Input.MOUSE_MODE_CAPTURED   # вернуть мышь камере (Esc её отпускал)
			get_viewport().set_input_as_handled()
			return
		if mb.button_index == MOUSE_BUTTON_RIGHT:
			_rmb = mb.pressed                       # блок; первые доли секунды — парирование
		elif mb.button_index == MOUSE_BUTTON_MIDDLE and mb.pressed:
			_toggle_lock()
		elif mb.button_index == MOUSE_BUTTON_LEFT and mb.pressed:
			_lmb_queued = true                   # удар; повтор во время удара — следующий удар комбо
		get_viewport().set_input_as_handled()   # колесо и щелчки — не камере игры под локацией
		return
	if event is InputEventMouseMotion:
		if Input.mouse_mode == Input.MOUSE_MODE_CAPTURED and _lock == null:
			_yaw -= (event as InputEventMouseMotion).relative.x * _f("turn_speed", 0.004) * _sens
			_cam_mouse = Time.get_ticks_msec()
		elif Input.mouse_mode == Input.MOUSE_MODE_CAPTURED and _lock != null:
			_lock_flick((event as InputEventMouseMotion).relative)
		return
	if event is InputEventKey and not _typing():
		var key: InputEventKey = event
		if key.physical_keycode in KEYS or key.keycode in KEYS:
			get_viewport().set_input_as_handled()   # горячие клавиши игры молчат, пока герой ходит
		if key.pressed and not key.echo and key.physical_keycode == KEY_E and is_instance_valid(_talk_npc) and not _near_gate:
			_start_talk()                            # житель рядом — разговор
		elif key.pressed and not key.echo and key.physical_keycode == KEY_E and _near_gate:
			exit_requested.emit()
		elif key.pressed and not key.echo and key.physical_keycode == KEY_E and is_instance_valid(_arena) \
				and _arena.can_start():
			_arena.start()
		if key.pressed and not key.echo and key.physical_keycode in [KEY_LEFT, KEY_RIGHT] and is_instance_valid(_arena) \
				and _arena.can_start():
			_arena.pick(-1 if key.physical_keycode == KEY_LEFT else 1)   # этап арены (с первого — до открытых)
			get_viewport().set_input_as_handled()
		if key.pressed and not key.echo and key.physical_keycode == KEY_V:
			_extra_try()                             # приём оружия в дополнительном слоте (катана — «Иай»)
		if key.pressed and not key.echo and key.physical_keycode == KEY_G:
			_ult_try()                               # ультимейт Разрубателя «Сокрушение»
		if key.pressed and not key.echo and key.physical_keycode == KEY_T:
			_extra_try()                             # то же, что V (по старой привычке)
		if key.pressed and not key.echo and key.physical_keycode == KEY_X and is_instance_valid(_arena):
			_arena.leave()                           # покинуть арену
		if key.pressed and not key.echo and key.physical_keycode == KEY_Q:
			_toggle_lock()                           # захват цели (как в Sekiro)
		if key.pressed and not key.echo and key.physical_keycode == KEY_Z:
			_shoulder = -_shoulder                   # плечо камеры в захвате
			_toast = {"text": mod.tr_key("pax_rpg_hud_shoulder_left" if _shoulder < 0.0 else "pax_rpg_hud_shoulder_right"),
				"t": 0.0}
		if key.pressed and not key.echo and key.physical_keycode == KEY_J:
			_set_journal(true)                       # журнал заданий
		if key.pressed and not key.echo and key.physical_keycode == KEY_M and is_instance_valid(_city_map):
			_set_city_map(true)                      # карта города (мир на паузе, мышь — карте)
		if key.pressed and not key.echo and key.physical_keycode == KEY_ESCAPE and is_instance_valid(_city_map) \
				and _city_map.visible:
			_city_map.visible = false
		elif key.pressed and not key.echo and key.physical_keycode == KEY_ESCAPE:
			_set_wrist(true)                         # меню — через наручный терминал
			get_viewport().set_input_as_handled()
		if key.pressed and not key.echo and key.physical_keycode == KEY_R:
			_drink()                                 # эликсир
		if key.pressed and not key.echo and key.physical_keycode == KEY_H:
			if is_instance_valid(_help):
				_help.set_open(not _help.is_open())   # окно «Управление»
		if key.pressed and not key.echo and key.physical_keycode == KEY_C and _hero != null:
			_hero.toggle_slot("back")                # снять / надеть плащ
		if key.pressed and not key.echo and (key.physical_keycode == KEY_F or key.keycode == KEY_F):
			_f_queued = true


var _yaw0 := INF                      # поворот тела в прошлом такте (было — мета узла: запись словаря каждый такт)


## Контакт с землёй: мягкое тёмное пятно под ногами (Decal, сверху вниз) — тает, когда герой в воздухе; и крен
## в повороте (на ходу и на бегу герой наклоняется внутрь дуги, как живой, а не едет по рельсам).
func _ground_contact(delta: float) -> void:
	if not is_instance_valid(_player) or not is_instance_valid(_body_mesh) or delta <= 0.0:
		return
	if _contact == null:
		_contact = Decal.new()
		var img := Image.create(64, 64, false, Image.FORMAT_RGBA8)
		for y: int in 64:
			for x: int in 64:
				var r := Vector2(x - 31.5, y - 31.5).length() / 31.5
				img.set_pixel(x, y, Color(0.0, 0.0, 0.0, clampf(1.0 - r, 0.0, 1.0) ** 1.6))
		_contact.texture_albedo = ImageTexture.create_from_image(img)
		_contact.size = Vector3(1.3, 1.6, 1.3)
		_contact.modulate = Color(0.0, 0.0, 0.0, 0.55)
		_contact.normal_fade = 0.4
		_contact.upper_fade = 0.2
		_contact.lower_fade = 0.6
		_view.add_child(_contact)
	var h := 0.9 if _hero == null else float(hero_cfg.get("capsule_height", 1.8)) * 0.5
	# стоит — пятно, его прозрачность и крен те же: не пишем (каждая запись — пересчёт трансформа / вызов рендера)
	var cp := _player.global_position - Vector3(0.0, h - 0.4, 0.0)
	if _contact.global_position != cp:
		_contact.global_position = cp
	var air := 0.0 if _player.is_on_floor() else 1.0
	var ca := lerpf(_contact.modulate.a, 0.55 * (1.0 - air), 1.0 - exp(-delta * 10.0))
	if absf(ca - _contact.modulate.a) > 1e-6:
		_contact.modulate.a = ca
	var hs := Vector2(_player.velocity.x, _player.velocity.z).length()
	var yaw := _body_mesh.rotation.y
	var turn := wrapf(yaw - (yaw if _yaw0 == INF else _yaw0), -PI, PI) / delta     # рад/с
	_yaw0 = yaw
	var busy := _hero != null and _hero.action != ""
	var want := 0.0 if busy or not _player.is_on_floor() else clampf(-turn * hs * _hf("lean", 0.012), -0.12, 0.12)
	_lean = lerpf(_lean, want, 1.0 - exp(-delta * 6.0))
	if absf(_body_mesh.rotation.z - _lean) > 1e-6:
		_body_mesh.rotation.z = _lean


## Пыль из-под ног: на земле и песке — заметное облачко, на камне — едва видное; на бегу — больше.
func _dust(at: Vector3, soft: bool, running: bool) -> void:
	if not bool(mod.get_setting("combat_fx", true)) and not soft:
		return
	if not soft and not running:
		return                                       # по камню шагом — без пыли (дёшево смотрелось)
	var p := CPUParticles3D.new()                    # (раньше узел создавался до этой проверки и на камне шагом утекал)
	p.one_shot = true
	p.explosiveness = 0.9
	p.amount = (6 if running else 3) if soft else 2
	p.lifetime = 1.1
	p.direction = Vector3(0.0, 1.0, 0.0)
	p.spread = 70.0
	p.initial_velocity_min = 0.2
	p.initial_velocity_max = 0.7
	p.gravity = Vector3(0.0, -0.4, 0.0)
	p.damping_min = 1.5
	p.damping_max = 2.5
	p.scale_amount_min = 0.08
	p.scale_amount_max = 0.18
	if not _fx_res.has("dmesh"):
		var sm := SphereMesh.new()
		sm.radius = 0.5
		sm.height = 1.0
		sm.radial_segments = 6
		sm.rings = 3
		var mat := StandardMaterial3D.new()
		mat.transparency = BaseMaterial3D.TRANSPARENCY_ALPHA
		mat.shading_mode = BaseMaterial3D.SHADING_MODE_UNSHADED
		mat.vertex_color_use_as_albedo = true
		sm.material = mat
		_fx_res["dmesh"] = sm
	p.mesh = _fx_res["dmesh"]
	var rk := "dramp%s" % soft
	if not _fx_res.has(rk):
		var ramp := Gradient.new()
		var c := Color(0.62, 0.55, 0.5) if soft else Color(0.6, 0.58, 0.6)
		ramp.set_color(0, Color(c, 0.0))
		ramp.add_point(0.15, Color(c, 0.09 if soft else 0.04))   # едва заметно
		ramp.set_color(ramp.get_point_count() - 1, Color(c, 0.0))
		_fx_res[rk] = ramp
	p.color_ramp = _fx_res[rk]
	var h := float(hero_cfg.get("capsule_height", 1.8)) * 0.5
	_view.add_child(p)
	p.global_position = at - Vector3(0.0, h - 0.05, 0.0)
	p.emitting = true
	get_tree().create_timer(1.5).timeout.connect(p.queue_free)
